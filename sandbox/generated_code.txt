#================================================================================
def twoSum(nums: list[int], target: int) -> list[int]:
    # Find the two indices that sum to the target
    for i in range(len(nums)):
        for j in range(i + 1, len(nums)):
            if nums[i] + nums[j] == target:
                return [i, j]
    return None

#Problem ID: isValid
#================================================================================
def isValid(s: str) -> bool:
    stack = []
    for c in s:
        if c in {'(','}':
            stack.append(c)
        elif c in {')','}':
            if len(stack) == 0 or stack[-1] != c:
                return False
            else:
                stack.pop()
    return len(stack) == 0

#Problem ID: mergeArrays
#================================================================================
def mergeArrays(a: list[int], b: list[int]) -> list[int]:
    result = []
    while a and b:
        if a[0] <= b[0]:
            result.append(a.pop(0))
        else:
            result.append(b.pop(0))
    return result

#Problem ID: maxSubArray
#================================================================================
def maxSubArray(nums: list[int]) -> int:
    return max(sum(nums[i:j]), i=0, j=len(nums)-1)

#================================================================================
#Problem ID: groupAnagrams
#================================================================================
def groupAnagrams(strs: list[str]) -> list[list[str]]:
    return sorted(set([str for strs in strs for _ in range(len(strs))]))

#Problem ID: mergeKLists
#================================================================================
def mergeKLists(lists: list[List[int]]) -> list[int]:
    # Create a temporary list to hold the merged list
    result = []
    
    # Iterate over each list in the input list
    for list in lists:
        # Append each element of the current list to the result list
        result.append(list)
        
    # Return the merged list
    return result

def trap(height: list[int]) -> int:
